function guardarConfig() {
    try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(config));
        const indicator = document.getElementById('saveText');
        if (indicator) {
            indicator.textContent = t('guardat_auto') + ' ✓';
            setTimeout(() => { indicator.textContent = t('guardat_auto'); }, 2000);
        }
        return true;
    } catch (e) { return false; }
}

function carregarConfig() {
    try {
        const dades = localStorage.getItem(STORAGE_KEY);
        if (dades) { config = { ...config, ...JSON.parse(dades) }; return true; }
    } catch (e) {}
    return false;
}

function aplicarConfigAlFormulari() {
    document.querySelectorAll('[data-campo]').forEach(el => {
        const campo = el.getAttribute('data-campo');
        if (config[campo] !== undefined) el.value = config[campo];
    });
    renderitzarCompetencies();
    renderitzarResultats();
    renderitzarSessions();
    renderitzarTemesColor();
}

function recollirDadesDelFormulari() {
    document.querySelectorAll('[data-campo]').forEach(el => {
        const campo = el.getAttribute('data-campo');
        config[campo] = el.value;
    });
}

function anarAlPas(num) {
    recollirDadesDelFormulari();
    guardarConfig();
    document.querySelectorAll('.pas').forEach(p => p.classList.remove('active'));
    document.getElementById('pas' + num).classList.add('active');
    document.querySelectorAll('.progress-step').forEach(s => {
        const pasNum = parseInt(s.getAttribute('data-pas'));
        s.classList.remove('active', 'completed');
        if (pasNum < num) s.classList.add('completed');
        else if (pasNum === num) s.classList.add('active');
    });
    pasActual = num;
    if (num === 5) { renderitzarResum(); actualitzarPreview(); }
    window.scrollTo({ top: 0, behavior: 'smooth' });
}

function seguentPas() { if (pasActual < 5) anarAlPas(pasActual + 1); }
function pasAnterior() { if (pasActual > 1) anarAlPas(pasActual - 1); }

function afegirCompetencia() { config.competencies.push({ codi: '', descripcio: '' }); renderitzarCompetencies(); guardarConfig(); }
function eliminarCompetencia(idx) { config.competencies.splice(idx, 1); renderitzarCompetencies(); guardarConfig(); }
function actualitzarCompetencia(idx, camp, valor) { config.competencies[idx][camp] = valor; guardarConfig(); }

function renderitzarCompetencies() {
    const container = document.getElementById('competenciesContainer');
    if (!container) return;
    if (config.competencies.length === 0) {
        container.innerHTML = '<p style="color:#999;font-style:italic;padding:10px;">' + t('btn_afegir_comp') + '</p>';
        return;
    }
    container.innerHTML = config.competencies.map((c, i) => 
        '<div class="competencia-item">' +
        '<button class="btn-remove" onclick="eliminarCompetencia(' + i + ')">×</button>' +
        '<div class="form-row">' +
        '<div class="form-group"><label>' + t('ph_comp_codi') + '</label><input type="text" value="' + c.codi + '" placeholder="CG1" onchange="actualitzarCompetencia(' + i + ', \'codi\', this.value)"></div>' +
        '<div class="form-group"><label>' + t('ph_comp_desc') + '</label><input type="text" value="' + c.descripcio + '" placeholder="..." onchange="actualitzarCompetencia(' + i + ', \'descripcio\', this.value)"></div>' +
        '</div></div>'
    ).join('');
}

function actualitzarResultat(idx, camp, valor) { config.resultats[idx][camp] = valor; guardarConfig(); }

function renderitzarResultats() {
    const container = document.getElementById('resultatsContainer');
    if (!container) return;
    if (config.resultats.length === 0) {
        container.innerHTML = '<p style="color:#999;font-style:italic;padding:10px;">' + t('btn_afegir_res') + '</p>';
        return;
    }
    container.innerHTML = config.resultats.map((r, i) => 
        '<div class="competencia-item">' +
        '<button class="btn-remove" onclick="eliminarResultat(' + i + ')">×</button>' +
        '<div class="form-row">' +
        '<div class="form-group"><label>' + t('ph_res_codi') + '</label><input type="text" value="' + r.codi + '" placeholder="C1" onchange="actualitzarResultat(' + i + ', \'codi\', this.value)"></div>' +
        '<div class="form-group"><label>' + t('ph_res_desc') + '</label><input type="text" value="' + r.descripcio + '" placeholder="..." onchange="actualitzarResultat(' + i + ', \'descripcio\', this.value)"></div>' +
        '</div></div>'
    ).join('');
}

function importarCSV(event) {
    const file = event.target.files[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = function(e) {
        try {
            const text = e.target.result;
            const linies = text.split(regexNL).filter(l => l.trim().length > 0);
            let inici = 0;
            if (linies.length > 0 && (linies[0].toLowerCase().includes('data') || linies[0].toLowerCase().includes('date') || linies[0].toLowerCase().includes('titol') || linies[0].toLowerCase().includes('title'))) {
                inici = 1;
            }
            let importades = 0;
            for (let i = inici; i < linies.length; i++) {
                const parts = linies[i].split(',').map(p => p.trim().replace(regexQuotes, ''));
                if (parts.length >= 2 && parts[0] && parts[1]) {
                    let data = parts[0];
                    if (data.match(/^\d{1,2}\/\d{1,2}\/\d{4}$/)) {
                        const [d, m, y] = data.split('/');
                        data = y + '-' + m.padStart(2,'0') + '-' + d.padStart(2,'0');
                    }
                    config.sessions.push({ data: data, titol: parts[1] });
                    importades++;
                }
            }
            renderitzarSessions();
            guardarConfig();
            mostrarToast(importades + ' ' + t('toast_csv_importat'), 'success');
        } catch (err) {
            console.error(err);
            mostrarToast(t('toast_csv_error'), 'error');
        }
    };
    reader.readAsText(file);
    event.target.value = '';
}

function actualitzarPreview() {

function exportarConfig() {
    recollirDadesDelFormulari();

        const blob = new Blob([JSON.stringify(config, null, 2)], { type: 'application/json' });

        const a = document.createElement('a');
        a.href = url;
        const nom = (config.nomCurt || config.nomAssignatura || 'configuracio').replace(regexSpaces, '_').toLowerCase();
        a.download = 'configuracio_' + nom + '_' + new Date().toISOString().split('T')[0] + '.json';
        document.body.appendChild(a); a.click(); document.body.removeChild(a);
        URL.revokeObjectURL(url);

}

function importarConfig(event) {
    const file = event.target.files[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = function(e) {
        try {
            const dades = JSON.parse(e.target.result);
            config = { ...config, ...dades };
            aplicarConfigAlFormulari();
            guardarConfig();
            mostrarToast(t('toast_importat'), 'success');
        } catch (err) { mostrarToast(t('toast_error_import'), 'error'); }
    };
    reader.readAsText(file);
    event.target.value = '';
}

function mostrarToast(msg, tipo) {
    tipo = tipo || 'info';
    const toast = document.createElement('div');
    toast.className = 'toast ' + tipo;
    toast.textContent = msg;
    document.body.appendChild(toast);
    setTimeout(() => { toast.style.transition = 'opacity 0.3s'; toast.style.opacity = '0'; setTimeout(() => toast.remove(), 300); }, 3000);
}

function formatData(dataStr) {
    if (!dataStr) return '';
    const parts = dataStr.split('-');
    if (parts.length !== 3) return dataStr;
    const y = parts[0], m = parts[1], d = parts[2];
    const meses_ca = ['gen','feb','mar','abr','mai','jun','jul','ago','set','oct','nov','des'];
    const meses_es = ['ene','feb','mar','abr','may','jun','jul','ago','sep','oct','nov','dic'];
    const meses_en = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
    const lang = localStorage.getItem('idioma_preferit') || 'ca';
    const meses = lang === 'es' ? meses_es : (lang === 'en' ? meses_en : meses_ca);
    return parseInt(d) + '-' + meses[parseInt(m)-1] + '-' + y;
}

    }
